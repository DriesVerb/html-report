Streaming SSE Chunk Wire Sequence

OpenAI streams the tool call name in the first chunk, then streams serialized JSON fragments in consecutive chunks until finish_reason: "tool_calls":

  "choices": [{

    "finish_reason": null
  }]
}

JS Engine Streaming Tool Accumulator

Production-Safe

Zero-dependency accumulation pattern that handles chunking, multiple simultaneous tools, and graceful JSON parse recovery: 

class OpenAIToolAccumulator {
  constructor() { this.tools = new Map(); }
  ingest(chunk) {
    const delta = chunk.choices?.[0]?.delta;
    if (!delta?.tool_calls) return null;
    for (const tc of delta.tool_calls) {
      const idx = tc.index ?? 0;
      if (!this.tools.has(idx)) {
        this.tools.set(idx, {
          id: tc.id || "",
          name: tc.function?.name || "",
          argsBuffer: tc.function?.arguments || ""
        });
      } else {
        const cur = this.tools.get(idx);
        if (tc.id) cur.id = tc.id;
        if (tc.function?.name) cur.name += tc.function.name;
        if (tc.function?.arguments) cur.argsBuffer += tc.function.arguments;
      }
    }
    if (chunk.choices?.[0]?.finish_reason === "tool_calls") {
      return Array.from(this.tools.values()).map(t => ({
        id: t.id,
        name: t.name,
        arguments: JSON.parse(t.argsBuffer || "{}")
      }));
    }
    return null;
  }
}

4. Turn & Session Termination Signals

Distinguishing between temporary model pauses (tool formulation) and definitive turn/session completion.

Termination FSM
finish_reason: "tool_calls" ACTIVE

The model paused because it requires local tool execution. The agent turn is NOT finished. Run the tool and return role: "tool".

finish_reason: "stop" COMPLETE

The model completed its natural language response or conversational turn. Hand prompt control back to the human developer.

finish_reason: "length" OVERFLOW

Generation stopped because max_completion_tokens or model context ceiling was reached. Flag token saturation error.

finish_reason: "content_filter" REJECTED

OpenAI safety moderation triggered. Prompt and generation halted. Log security telemetry event.

import OpenAI from "openai";
const openai = new OpenAI();
const runStream = openai.beta.threads.runs.stream(threadId, { assistant_id });
// 1. Tool Call Initiated
runStream.on("toolCallCreated", (toolCall) => {
  telemetry.emit("tool_start", { id: toolCall.id, name: toolCall.function?.name });
});
// 2. Argument Buffer Stream
runStream.on("toolCallDelta", (delta) => {
  telemetry.appendBuffer(delta.index, delta.function?.arguments);
});
// 3. Human Approval Requested
runStream.on("event", (evt) => {
  if (evt.event === "thread.run.requires_action") {
    telemetry.emit("waiting_user", evt.data.required_action);
  }
});

async function listenChatSSE(apiKey, payload, onChunk) {
  const res = await fetch("https://api.openai.com/v1/chat/completions", {
    method: "POST",
    headers: { "Authorization": `Bearer ${apiKey}`, "Content-Type": "application/json" },
    body: JSON.stringify({ ...payload, stream: true })
  });
  const reader = res.body.getReader();
  const decoder = new TextDecoder();
  let buffer = "";
  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    buffer += decoder.decode(value, { stream: true });
    const lines = buffer.split("\n");
    buffer = lines.pop();
    for (const l of lines) {
      if (l.startsWith("data: ") && !l.includes("[DONE]")) {
        onChunk(JSON.parse(l.slice(6)));
      }
